/*
long is used here because of a overflow that can happen when n is large
Gauss formula can find our missing num but i did not think of that until after
 */

import java.util.Arrays;

public static int findMissingNumber(int[] numbers) {
    int[] sorted = numbers.clone();   // copy
    Arrays.sort(sorted);

    // Missing the first number
    if (sorted.length == 0 || sorted[0] != 1) {
        return 1;
    }

    //  if the next number isn't current + 1, current + 1 is missing
    for (int i = 0; i < sorted.length - 1; i++) {
        if (sorted[i] + 1 != sorted[i + 1]) {
            return sorted[i] + 1;
        }
    }

    // No gap anywhere, so the missing number is the one after the last
    return sorted[sorted.length - 1] + 1;
}
