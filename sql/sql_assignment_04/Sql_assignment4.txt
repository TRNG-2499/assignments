CREATE TABLE Department(
department_id INTEGER PRIMARY KEY,
department_name VARCHAR(200),
location VARCHAR(200)
);

CREATE TABLE Doctor(
doctor_id INTEGER PRIMARY KEY,
department_id INTEGER,
first_name VARCHAR(100),
last_name VARCHAR(100),
specialty VARCHAR(100),
phone VARCHAR(100),

CONSTRAINT department_id_fk FOREIGN KEY (department_id) REFERENCES Department(department_id)
);

CREATE TABLE Room(
room_id INTEGER PRIMARY KEY,
department_id INTEGER,
room_number VARCHAR(100),
room_type VARCHAR(100),
is_available BOOLEAN,

CONSTRAINT department_id_fk2 FOREIGN KEY (department_id) REFERENCES Department(department_id)
);

CREATE TABLE Patient(
patient_id INTEGER PRIMARY KEY,
first_name VARCHAR(100),
last_name VARCHAR(100),
date_of_birth DATE,
phone VARCHAR(100),
address VARCHAR(200)
);

CREATE TABLE Appointment(
appointment_id INTEGER PRIMARY KEY,
patient_id INTEGER,
doctor_id INTEGER,
appointment_date DATE,
appointment_time TIME,
reason VARCHAR(200),
status VARCHAR(100),

CONSTRAINT patient_id_fk FOREIGN KEY (patient_id) REFERENCES Patient(patient_id),
CONSTRAINT doctor_id_fk FOREIGN KEY (doctor_id) REFERENCES Doctor(doctor_id)
);

CREATE TABLE Prescription(
prescription_id INTEGER PRIMARY KEY,
appointment_id INTEGER,
medication_name VARCHAR(100),
dosage VARCHAR(100),
instructions VARCHAR(400),

CONSTRAINT appointment_id_fk2 FOREIGN KEY (appointment_id) REFERENCES Appointment(appointment_id)
);
