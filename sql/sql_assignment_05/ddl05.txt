
Principles have a name, hire date, and salary.
CREATE TABLE principle (
    principle_id INT PRIMARY KEY,
    name VARCHAR(255),
    hire_date DATE,
    SALARY int
)

Schools have a name, and a single principle
CREATE TABLE school (
    school_id INT PRIMARY KEY,
    name VARCHAR(255),
    principle_id INT REFERENCES principle(principle_id)
)

Students have a name, a grade level (7th, 8th, 9th, etc...), a home address, an emergency contact phone number, and a school
CREATE TABLE students (
    student_id INT PRIMARY KEY,
    name VARCHAR(100),
    grade TINYINT,
    address VARCHAR(255),
    number INT,
    school_id INT REFERENCES school(school_id)
)
