CREATE VIEW ProductMargin AS
SELECT product_id, category_id, product_name, unit_cost, unit_price, stock_quantity,
    unit_price - unit_cost AS profit_margin
FROM Product;

CREATE VIEW ProductSalesSummary AS
SELECT p.product_id, p.product_name,
    SUM(s.quantity_sold) AS total_units_sold,
    SUM(s.quantity_sold * p.unit_price) AS total_revenue,
    SUM(s.quantity_sold * p.unit_cost) AS total_cost,
    SUM(s.quantity_sold * (p.unit_price - p.unit_cost)) AS total_profit
FROM Product p
JOIN Sale s ON s.product_id = p.product_id
GROUP BY p.product_id, p.product_name;


SELECT product_name, unit_cost, unit_price, profit_margin
FROM ProductMargin;

SELECT c.category_name, SUM(p.unit_cost * p.stock_quantity) AS total_inventory_value
FROM Category c
JOIN Product p ON p.category_id = c.category_id
GROUP BY c.category_id, c.category_name;

SELECT product_name, unit_price * stock_quantity AS gross_revenue_potential
FROM Product;

SELECT AVG(profit_margin) AS average_profit_margin
FROM ProductMargin;

SELECT product_name, profit_margin
FROM ProductMargin
WHERE profit_margin > (SELECT AVG(profit_margin) FROM ProductMargin);

SELECT product_name, total_units_sold, total_revenue, total_cost, total_profit
FROM ProductSalesSummary;

SELECT product_name, total_profit
FROM ProductSalesSummary
ORDER BY total_profit DESC
LIMIT 3;

SELECT product_name
FROM Product
WHERE LENGTH(product_name) = (SELECT MAX(LENGTH(product_name)) FROM Product);

SELECT product_name
FROM Product
WHERE LENGTH(product_name) = (SELECT MIN(LENGTH(product_name)) FROM Product);

SELECT product_name
FROM Product
WHERE product_name LIKE '% %';

SELECT category_name
FROM Category
WHERE category_name LIKE 'E%';

SELECT UPPER(p.product_name) AS product_name, LOWER(c.category_name) AS category_name
FROM Product p
JOIN Category c ON c.category_id = p.category_id;
