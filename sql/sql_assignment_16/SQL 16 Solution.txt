SELECT d.first_name, d.last_name
FROM Doctor d
LEFT JOIN Appointment a ON a.doctor_id = d.doctor_id
WHERE a.appointment_id IS NULL;


-- Stretch Goal:

SELECT d.first_name, d.last_name
FROM Doctor d
LEFT JOIN Appointment a ON a.doctor_id = d.doctor_id AND a.status = 'Scheduled'
WHERE a.appointment_id IS NULL;
